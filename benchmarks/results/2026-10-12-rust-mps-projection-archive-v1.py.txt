import datetime
import hashlib
import json
from pathlib import Path
import shutil
import subprocess

root = Path.cwd()
def digest(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream,'sha256').hexdigest()
def query(args):
    p=subprocess.run(args,capture_output=True,text=True,timeout=30)
    return {'command':args,'exit_code':p.returncode,'stdout':p.stdout,'stderr':p.stderr}
pids=[51172,51212,51226,43111,43153,43161,96097,96112,96121,17694,85019,85027,88184,35939]
absence=query(['ps','-p',','.join(map(str,pids)),'-o','pid='])
assert absence['exit_code']==1 and not absence['stdout'].strip() and not absence['stderr'].strip()
vm=query(['/opt/podman/bin/podman','machine','inspect','podman-machine-default'])
assert vm['exit_code']==0
machine=json.loads(vm['stdout'])[0]
assert machine['State']=='stopped'
previous=json.loads(Path('benchmarks/results/2026-10-12-metal-gating-tails-idle-v1.json').read_text())
protected=previous['protected_sha256']
for name,sha in protected.items(): assert digest(name)==sha
outputs={}
for stem in ['2026-10-12-rust-mps-projection-v3','2026-10-12-rust-mps-projection-v1','2026-10-12-rust-mps-projection-v2']:
    record=json.loads(Path(f'benchmarks/raw/{stem}.json').read_text())
    if stem=='2026-10-12-rust-mps-projection-v1':
        assert record['completed'] is False and record['failure']['stage']=='build_checks'
        assert record['native_runs']==[]
    else:
        assert record['completed'] is True and 'failure' not in record
    for suffix in ['.json','.json.stderr.txt']:
        source=Path(f'benchmarks/raw/{stem}{suffix}')
        target=Path(f'benchmarks/results/{stem}{suffix}')
        with target.open('xb') as output: output.write(source.read_bytes())
        assert digest(source)==digest(target)
        outputs[str(target)]=digest(target)
# Preserve changed copied Rust sources for failed/superseded attempts.
for label,build in [('build-failed','rust-mps-projection-v1'),('superseded','rust-mps-projection-v2')]:
    source=Path(f'artifacts/{build}/src/rust_metal_projection.rs')
    target=Path(f'benchmarks/results/2026-10-12-rust-mps-projection-{label}.rs.txt')
    with target.open('xb') as output: output.write(source.read_bytes())
    outputs[str(target)]=digest(target)
script=Path('artifacts/archive-rust-mps-projection-v1.py')
target=Path('benchmarks/results/2026-10-12-rust-mps-projection-archive-v1.py.txt')
with target.open('xb') as output: output.write(script.read_bytes())
outputs[str(target)]=digest(target)

       'previous_idle_sha256':digest('benchmarks/results/2026-10-12-metal-gating-tails-idle-v1.json'),
