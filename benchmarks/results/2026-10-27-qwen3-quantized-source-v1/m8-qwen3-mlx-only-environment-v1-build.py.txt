"""Create an owned offline MLX-only venv from exact installed package bytes."""
from pathlib import Path
import hashlib,importlib.metadata,json,os,shutil,signal,sys,sysconfig,venv
R=Path.cwd();O=Path(__file__).resolve().parent;sys.path.insert(0,str(R/'tools/reference'));import qwen3_quantized_source as source
OLD=Path(sysconfig.get_path('purelib')).resolve();DEST=R/'.venv/mlx-source-only-v1';roots=set(source.PACKAGES)|{'mlx','numpy','tokenizers','jinja2','markupsafe'};roots.update(p.name for p in OLD.glob('numpy*.libs'))
def ident(s):return s.st_dev,s.st_ino,s.st_mode,s.st_size,s.st_mtime_ns,s.st_ctime_ns
def pin(path):
 assert path.is_file()and not path.is_symlink(),path;before=path.stat();h=hashlib.sha256()
 with path.open('rb')as f:
  while body:=f.read(65536):h.update(body)
 assert ident(before)==ident(path.stat()),path
 return dict(bytes=before.st_size,sha256=h.hexdigest())
def inventory():
 selected={OLD/name for name in roots if(OLD/name).exists()};distributions=[]
 for d in importlib.metadata.distributions(path=[str(OLD)]):
  files=d.files or[];provided={Path(str(f)).parts[0]for f in files if Path(str(f)).parts}
  if provided&roots:
   metadata=Path(d._path);assert metadata.parent==OLD and metadata.suffix=='.dist-info';selected.add(metadata);distributions.append(dict(name=d.metadata['Name'],version=d.version,metadata=metadata.name))
 assert not any(d['name'].lower().replace('_','-')in('torch','torchvision','torchaudio','torchgen')for d in distributions)
 result={}
 for root in sorted(selected):
  assert not root.is_symlink()
  paths=[root]if root.is_file()else sorted(root.rglob('*'))
  for p in paths:
   if '__pycache__'in p.parts:continue
   assert not p.is_symlink(),p
   if p.is_file():
    assert p.suffix!='.pth',p
    result[str(p.relative_to(OLD))]=pin(p)
 distributions.sort(key=lambda d:(d['name'].lower(),d['version'],d['metadata']))
 return result,distributions
def stop(number,_):
 for sig in(signal.SIGINT,signal.SIGTERM):signal.signal(sig,signal.SIG_IGN)
 raise InterruptedError('environment copy signal '+str(number))
def main():

  base=Path(sys._base_executable).resolve();invoking=Path(sys.executable).resolve();base_pin=pin(base);invoking_pin=pin(invoking)

  assert not list(site.glob('*.pth'))and not(site/'torch').exists()and not list(site.glob('torch*.dist-info'))
