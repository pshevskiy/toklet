"""Verify exact owned natural exits and archive completed scalar application data."""
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import subprocess


def digest(path):
    with Path(path).open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


owned_path = Path("artifacts/2026-10-11-application-scalar-owned-processes-v1.json")
owned = json.loads(owned_path.read_text())
completion_path = Path("artifacts/2026-10-11-application-scalar-session-completion-v1.json")
completion = json.loads(completion_path.read_text())
assert completion["session_id"] == 16633 and completion["exit_code"] == 0
assert "Full scalar application transport completed naturally" in completion["final_output"]
assert completion["origin"] == "Observed unified exec session completion; not inferred from result data or PID absence"
pids = [owned[key] for key in ("launcher_pid", "transport_pid", "worker_pid")]
assert pids == [65037, 65046, 65060]
snapshot = subprocess.run(["ps", "-p", ",".join(map(str, pids)),
                           "-o", "pid=,ppid=,stat=,command="], capture_output=True, text=True)
assert snapshot.returncode == 1 and not snapshot.stdout.strip() and not snapshot.stderr.strip()
held = subprocess.check_output(["ps", "-p", "17694,85019", "-o", "pid=,ppid=,stat=,command="], text=True)
rows = {int(row[0]): row for line in held.splitlines() if len(row := line.split(None, 3)) == 4}
assert rows[17694][1] == "35939" and "T" in rows[17694][2] and rows[17694][3].endswith("/Python -")
assert rows[85019][1] == "17694" and rows[85019][2].startswith("Z")
base = "benchmarks/raw/2026-10-11-application-scalar"
result = json.loads(Path(base + "-v1.json").read_text())
samples = [json.loads(line) for line in Path(base + "-v1.json.samples.jsonl").read_text().splitlines()]
assert result["qualification"] == "three_warmups_ten_samples"
assert result["setup"]["warmup_repetitions"] == 3 and len(samples) == 10
assert samples == result["samples"] and all(len(sample["turns"]) == 3 for sample in samples)
launch = json.loads(Path(base + "-launch-v1.json").read_text())
assert launch["launcher_pid"] == owned["launcher_pid"]
assert launch["launch_utc"] < owned["utc"]
assert digest("tools/bench/application.py") == launch["driver_sha256"]
assert digest("artifacts/launch-scalar-application-2026-10-11-v1.py") == launch["producer_sha256"]
end_verified = {}
for section in ("frozen_binaries", "validated_local_model_assets"):
    end_verified[section] = {}
    for name, expected in launch["verified"][section].items():
        assert digest(name) == expected, name
        end_verified[section][name] = expected
copies = {
    base + "-v1.json": "benchmarks/results/2026-10-11-application-scalar-v1.json",
    base + "-v1.json.samples.jsonl": "benchmarks/results/2026-10-11-application-scalar-v1.json.samples.jsonl",
    base + "-launch-v1.json": "benchmarks/results/2026-10-11-application-scalar-launch-v1.json",
    str(owned_path): "benchmarks/results/2026-10-11-application-scalar-owned-processes-v1.json",
    "tools/bench/application.py": "benchmarks/results/2026-10-11-application-scalar-driver-v1.py.txt",
    "artifacts/launch-scalar-application-2026-10-11-v1.py": "benchmarks/results/2026-10-11-application-scalar-launch-v1.py.txt",
    str(completion_path): "benchmarks/results/2026-10-11-application-scalar-session-completion-v1.json",
    __file__: "benchmarks/results/2026-10-11-application-scalar-idle-v1.py.txt",
}
assert all(not Path(target).exists() for target in copies.values())
saved = {}
for source, target in copies.items():
    with Path(target).open("xb") as stream:
        stream.write(Path(source).read_bytes())
    saved[target] = digest(target)
receipt = {
    "format": "toklet-scalar-application-natural-exit-idle-v1",
    "idle_verified_utc": datetime.now(timezone.utc).isoformat(),
    "completed_workload": "application", "measured_scenarios": 10, "measured_turns": 30,
    "warmup_scenarios": 3, "uninterrupted_batch": True,
    "exited_launcher_pid": pids[0], "exited_transport_pid": pids[1], "exited_numerical_worker_pid": pids[2],
    "held_original_orchestration_pid": 17694, "snapshot": snapshot.stdout, "held_snapshot": held,
    "saved_result_sha256": saved, "producer_sha256": digest(__file__),
    "session_completion_sha256": digest(completion_path),
    "end_verified": end_verified,
}
with Path("benchmarks/results/2026-10-11-application-scalar-idle-v1.json").open("x") as stream:
    json.dump(receipt, stream, indent=2, allow_nan=False)
    stream.write("\n")
print(json.dumps(receipt))
