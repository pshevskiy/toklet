def digest(path):
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()

def checked(name, expected):
    value = digest(root / name)
    assert value == expected, (name, value, expected)
    return value

def read(name):
    return json.loads((root / name).read_text())

def utc():
    return datetime.now(timezone.utc).isoformat()

def snapshot():
    return subprocess.check_output([
        "ps", "-o", "pid,ppid,stat,time,command", "-p", "17694,80030,80039",
    ], text=True)

def assert_idle(value):
    rows = {int(row[0]): row for line in value.splitlines()[1:]
            if len(row := line.split(None, 4)) == 5}
    assert 80039 not in rows
    assert int(rows[17694][1]) == 35939 and "T" in rows[17694][2]
    assert int(rows[80030][1]) == 17694 and rows[80030][2].startswith("Z")
