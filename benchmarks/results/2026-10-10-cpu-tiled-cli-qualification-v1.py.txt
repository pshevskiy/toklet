def checked(command):
    clock=time.perf_counter();process=subprocess.Popen(command,cwd=ROOT,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True,start_new_session=True)
    timed_out=False
    try:
        stdout,stderr=process.communicate(timeout=1200)
    except subprocess.TimeoutExpired:
        timed_out=True
        try:os.killpg(process.pid,signal.SIGKILL)
        except ProcessLookupError:pass
        stdout,stderr=process.communicate(timeout=20)
    result['checks'].append({'command':command,'exit_code':process.returncode,'stdout':stdout,'stderr':stderr,'timed_out':timed_out,'wall_seconds':time.perf_counter()-clock})
    if process.returncode or timed_out:raise ValueError(f'Check failed: {command}')

def semantic(event):
    if event['event']=='done':
        return {'event':'done','generation':{name:value for name,value in event['statistics']['generation'].items() if not name.endswith('_seconds')}}
    return {name:value for name,value in event.items() if name not in ('elapsed_seconds','cpu_projection')}

def run_cli(binary,role,projection):
    flags=[] if role in ('frozen_scalar','default') else ['--cpu-projection',projection]
    command=[str(binary),'--model',str(ROOT/'models/Qwen2.5-0.5B-Instruct'),'--interactive','--json','--context-capacity','2048','--prefill-chunk','128','--no-eos']+flags
    raw=OUT.with_suffix(OUT.suffix+f'.{role}.jsonl');stderr=OUT.with_suffix(OUT.suffix+f'.{role}.stderr.txt')
    if raw.exists() or stderr.exists():raise ValueError('CLI raw evidence already exists')
    events=[];cases=[]
    with tempfile.TemporaryFile(mode='w+t') as errors,raw.open('x') as evidence:
        process=subprocess.Popen(command,cwd=ROOT,stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=errors,text=True)
        watchdog=threading.Timer(1800,process.kill);watchdog.start()
        try:
            def read():
                line=process.stdout.readline()
                if not line:raise ValueError('Unexpected CLI EOF')
                event=json.loads(line);events.append(event);evidence.write(line);evidence.flush();return event
            def send(value):process.stdin.write(json.dumps(value)+'\n');process.stdin.flush()
            ready=read();assert ready['event']=='ready' and ready['backend']=='cpu' and ready['context_capacity']==2048
            if role!='frozen_scalar':assert ready['cpu_projection']==projection
            assert ready['memory_admitted_bytes']<=6*1024**3
            for name in ('future_changed','short','single'):
                send({'tokens':fixture['cases'][name]['tokens'],'max_new_tokens':1})
                generated=[];semantic_events=[]
                for action in ('generate','continue'):
                    if action=='continue':send({'action':'continue','max_new_tokens':1})
                    ids=[]
                    while True:
                        event=read();semantic_events.append(semantic(event))
                        if event['event']=='token':ids.append(event['token_id'])
                        elif event['event']=='done':
                            generation=event['statistics']['generation']
                            assert generation['reason']=={'kind':'output_limit'} and generation['sampled_tokens']==1 and generation['emitted_tokens']==1 and generation['pending_tokens']==1
                            if role!='frozen_scalar':assert event['statistics']['cpu_projection']==projection
                            if action=='generate':assert generation['prefill_tokens']==len(fixture['cases'][name]['tokens']) and generation['reused_tokens']==0
                            else:assert generation['prefill_tokens']==1 and generation['reused_tokens']==len(fixture['cases'][name]['tokens'])
                            break
                        elif event['event']!='text':raise ValueError(event)
                    assert len(ids)==1;generated+=ids
                assert generated[0]==fixture['argmax'][name][-1]
                send({'action':'reset'})
                while True:
                    event=read();semantic_events.append(semantic(event))
                    if event['event']=='reset':break
                    assert event['event']=='text'
                cases.append({'name':name,'tokens':fixture['cases'][name]['tokens'],'expected_first_token':fixture['argmax'][name][-1],
                              'token_ids':generated,'semantic_events':semantic_events})
            peak=common.footprint_peak(process.pid);assert peak is not None and 0<peak<=6*1024**3
            process.stdin.close();assert process.wait(timeout=20)==0 and not process.stdout.read().strip()
        finally:
            watchdog.cancel()
            if process.poll() is None:process.kill()
            process.wait(timeout=20);process.stdout.close()
            if not process.stdin.closed:process.stdin.close()
            errors.seek(0);stderr.write_text(errors.read());evidence.flush()
            result.setdefault('partial_runs',{})[role]={'command':command,'exit_code':process.returncode,
                'raw':str(raw.relative_to(ROOT)),'raw_sha256':prepare.digest(raw),
                'stderr':str(stderr.relative_to(ROOT)),'stderr_sha256':prepare.digest(stderr)}
    return {'passed':True,'cpu_projection':projection,'command':command,'ready':ready,'cases':cases,
            'physical_process_charge_peak_bytes':peak,'exit_code':process.returncode,
            'raw':str(raw.relative_to(ROOT)),'raw_sha256':prepare.digest(raw),'stderr_sha256':prepare.digest(stderr)}
