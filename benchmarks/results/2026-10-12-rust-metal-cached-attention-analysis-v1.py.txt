inputs,anchors=cached.cases();full,fa=attention.cases();ci,ca=cache.cases();ri,ra,_=rotary.cases();mi,ma,_=composed.cases()
validators={f'cached_{i+1}':lambda e,c=c:attention.validate(e,c,anchors[c['id']]) for i,c in enumerate(inputs)}

f32=lambda v:struct.unpack('<f',struct.pack('<f',v))[0]
stages={name:{'first_frame_values':0,'equal_frozen_bits':0,'max_source_fp32_abs_error':0.,'max_source_fp32_gate_fraction':0.,'first_frame_guard_values':0} for name in ('scores','probabilities','context')}

    r=raw[3+i]['result'];rows=item['rows'];valid=item['valid'];offset=item['offset'];width=r['threadgroup_width'];units=math.ceil(rows*4/width)*width

    q,k,v=[struct.unpack(f'<{len(item[key])}e',struct.pack(f'<{len(item[key])}H',*item[key])) for key in ('q_bits','k_bits','v_bits')]
