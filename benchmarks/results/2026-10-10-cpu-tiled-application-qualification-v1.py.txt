def checked(command):
    clock=time.perf_counter();process=subprocess.Popen(command,cwd=ROOT,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True,start_new_session=True)
    timed_out=False
    try:stdout,stderr=process.communicate(timeout=1200)
    except subprocess.TimeoutExpired:
        timed_out=True
        try:os.killpg(process.pid,signal.SIGKILL)
        except ProcessLookupError:pass
        stdout,stderr=process.communicate(timeout=20)
    record['checks'].append({'command':command,'exit_code':process.returncode,'stdout':stdout,'stderr':stderr,
                             'timed_out':timed_out,'wall_seconds':time.perf_counter()-clock})
    if process.returncode or timed_out:raise ValueError(f'Check failed: {command}')

def semantics(result):
    assert result['qualification']=='functional_probe' and result['setup']['warmup_repetitions']==0 and len(result['samples'])==1
    turns=result['samples'][0]['turns'];assert len(turns)==3
    state=[]
    for turn in turns:
        generation={key:value for key,value in turn['statistics']['generation'].items() if not key.endswith('_seconds')}
        assert generation['emitted_tokens']==len(turn['token_ids'])==2 and generation['pending_tokens']==1
        assert generation['reason']=={'kind':'output_limit'} and generation['reused_tokens']==0
        state.append({'token_ids':turn['token_ids'],'visible_text':turn['visible_text'],
                      'visible_text_sha256':turn['visible_text_sha256'],'generation':generation})
    return state

    checked([str(ROOT/'.venv/bin/python'),'-m','unittest','discover','-s','tools/bench','-p','test_*.py'])
    probes={}
    for role,projection in (('legacy','scalar'),('scalar','scalar'),('tiled','tiled')):
        output=ROOT/f'benchmarks/raw/2026-10-10-application-adapter-{role}-probe-v1.json'

        result=prepare.load_json(output)
        assert result['identity']['cpu_projection']==projection
        assert result['identity']['binary_sha256']==(FROZEN['target/release/toklet'] if role=='legacy' else cli['binary_sha256'])
        assert result['identity']['cli_qualification_sha256']==(None if role=='legacy' else prepare.digest(CLI))
        assert 0<result['memory']['admitted_bytes']<=6*1024**3 and 0<result['memory']['peak_process_bytes']<=6*1024**3
        if role!='legacy':assert all(turn['statistics']['cpu_projection']==projection for turn in result['samples'][0]['turns'])
        raw=output.with_suffix(output.suffix+'.samples.jsonl')
        assert [json.loads(line) for line in raw.read_text().splitlines()]==result['samples']
        probes[role]={'path':str(output.relative_to(ROOT)),'sha256':prepare.digest(output),

                      'semantics':semantics(result),'memory':result['memory']}
        assert probes[role]['semantics']==probes['legacy']['semantics']
    assert probes['scalar']['memory']['admitted_bytes']==probes['tiled']['memory']['admitted_bytes']
    assert probes['scalar']['memory']['logical_allocations']==probes['tiled']['memory']['logical_allocations']
    record['probes']=probes
