def evaluate(number, arguments=(), architecture=0xC00000B7):
    """Independent classic-BPF interpreter over the documented seccomp_data ABI."""
    data = struct.pack("<iIQ6Q", number, architecture, 0, *(list(arguments) + [0] * (6 - len(arguments))))
    program = child.filter_instructions()
    accumulator, pc = 0, 0
    for _ in range(len(program) + 1):
        code, yes, no, value = program[pc]
        if code == 0x20:
            accumulator = struct.unpack_from("<I", data, value)[0]
        elif code in (0x15, 0x45):
            condition = accumulator == value if code == 0x15 else bool(accumulator & value)
            pc += yes if condition else no
        elif code == 0x06:
            return value
        else:
            raise AssertionError("Unexpected classic BPF instruction")
        pc += 1
        if not 0 <= pc < len(program):
            raise AssertionError("Classic BPF branch escapes its fixed program")
    raise AssertionError("Classic BPF did not terminate")


def ready(mode):
    blob = b"".join(struct.pack("<HBBI", *row) for row in child.filter_instructions())
    return {"event": "isolated", "mode": mode, "policy": child.POLICY.copy(),
            "profile": "toklet-linux-aarch64-seccomp-v1",
            "limits": {name: [value, value] for name, (_, value) in child.limits().items()},
            "runtime": {"machine": "aarch64", "kernel": "authored-fixture",
                        "python_executable": "/usr/local/bin/python3", "python_sha256": "1" * 64,
                        "uid": 65534, "gid": 65534, "filter_sha256": hashlib.sha256(blob).hexdigest()},
            "python": "offline fixture", "isolated_flag": 1, "no_site_flag": 1,
            "optimize_flag": 0, "dont_write_bytecode": True,
            "environment_keys": sorted(probe.ENVIRONMENT)}


def capture(mode="benign", result=None, container_code=0, **updates):
    payload = (json.dumps(ready(mode)) + "\n").encode()
    if result is not None:
        payload += (json.dumps({"event": "probe_result", "mode": mode, "result": result}) + "\n").encode()
    record = {"mode": mode, "exit_code": -signal.SIGKILL if container_code == 137 else container_code,
              "container_exit_code": container_code, "transport_exit_code": container_code,
              "container_state": {"Running": False, "OOMKilled": False, "Error": "", "ExitCode": container_code},
              "container_removed": True, "container_cleanup_errors": [],
              "container_forced_kill": False, "container_timeout_kill": False,
              "execution_killed": False, "process_group_killed": False,
              "wall_timed_out": False, "output_limit_exceeded": False,
              "elapsed_seconds": 0.1, "cleanup_errors": [], "surviving_group_members": []}
    for name, content in [("stdout", payload), ("stderr", b"")]:
        record[name + "_base64"] = base64.b64encode(content).decode()
        record[name + "_captured_bytes"] = len(content)
        record[name + "_captured_sha256"] = hashlib.sha256(content).hexdigest()
    record.update(updates)
    return record

    def test_api_connection_must_identify_the_fixed_rootless_machine(self):
        machine = {"Name": "podman-machine-default", "Rootful": False,
                   "SSHConfig": {"Port": 51387, "RemoteUsername": "core", "IdentityPath": "/fixed/machine/key"}}
        connection = {"Name": "podman-machine-default", "URI": "ssh://core@127.0.0.1:51387/run/user/1000/podman/podman.sock", "Identity": "/fixed/machine/key", "Default": False}
        bound = probe.bound_machine_connection(machine, [connection, {"Name": "unrelated", "Default": True}])
        self.assertEqual(bound["URI"], connection["URI"])
        for change in [{"URI": connection["URI"].replace("51387", "51388")},
                       {"URI": connection["URI"].replace("127.0.0.1", "elsewhere")},
                       {"URI": connection["URI"].replace("core@", "root@")},
                       {"URI": "unix:///foreign/socket"}, {"Identity": "/another/key"}, {"Name": "other"}]:
            with self.subTest(change=change), self.assertRaises(ValueError):
                probe.bound_machine_connection(machine, [{**connection, **change}])
        for change in [{"Rootful": True}, {"Name": "other"}, {"SSHConfig": {**machine["SSHConfig"], "Port": True}}]:
            with self.assertRaises(ValueError): probe.bound_machine_connection({**machine, **change}, [connection])
        with self.assertRaises(ValueError): probe.bound_machine_connection(machine, [connection, connection])

    def test_architecture_guard_kills_wrong_abi_before_syscall_selection(self):
        for architecture in [0xC000003E, 0x40000028, 0x800000B7, 0]:
            for number in [64, 93, 222, -1]:
                self.assertEqual(evaluate(number, architecture=architecture), 0x80000000)

    def test_default_denies_files_network_processes_and_limit_changes(self):
        # Actual AArch64 syscall numbers from the independent UAPI inventory.
        for number in [56, 437, 78, 35, 198, 203, 200, 220, 435, 221, 281,
                       167, 277, 261, 164, 103, 117, 270, 271, -1, 9999]:
            self.assertEqual(evaluate(number), 0x00050000 | errno.EPERM)
        for number in [93, 94, 139, 214, 215, 216, 233, 113, 114, 115, 101, 98, 124, 172, 178, 278]:
            self.assertEqual(evaluate(number), 0x7FFF0000)

    def test_descriptor_guards_reject_other_fds_and_poisoned_high_bits(self):
        for number, allowed in [(63, {0}), (64, {1, 2}), (57, {0, 1, 2}), (80, {0, 1, 2})]:
            for descriptor in range(6):
                self.assertEqual(evaluate(number, [descriptor]), 0x7FFF0000 if descriptor in allowed else 0x00050001)
            for descriptor in [1 << 32, (1 << 32) + 1, (1 << 32) + 2, 2**64 - 1]:
                self.assertEqual(evaluate(number, [descriptor]), 0x00050001)

    def test_memory_mapping_requires_anonymous_non_executable_arguments(self):
        good = [0, 4096, 3, 34, 2**64 - 1, 0]
        self.assertEqual(evaluate(222, good), 0x7FFF0000)
        self.assertEqual(evaluate(222, [0, 2**31, 3, 33, 2**64 - 1, 0]), 0x7FFF0000)
        for index, value in [(2, 7), (2, 3 + 2**32), (3, 2), (3, 34 + 2**32), (4, 0), (4, 3)]:
            changed = good.copy()
            changed[index] = value
            self.assertEqual(evaluate(222, changed), 0x00050001)
        self.assertEqual(evaluate(226, [0, 4096, 3]), 0x7FFF0000)
        for protection in [4, 7, 3 + 2**32]:
            self.assertEqual(evaluate(226, [0, 4096, protection]), 0x00050001)

    def test_ctypes_instruction_layout_matches_independent_uapi_packing(self):
        rows = child.filter_instructions()
        self.assertLessEqual(len(rows), 255)
        array = (child.SockFilter * len(rows))(*(child.SockFilter(*row) for row in rows))
        self.assertEqual(bytes(array), b"".join(struct.pack("<HBBI", *row) for row in rows))
        self.assertEqual((ctypes.sizeof(child.SockFilter), child.SockFilter.k.offset), (8, 4))
        self.assertEqual((ctypes.sizeof(child.SockFprog), child.SockFprog.filter.offset), (16, 8))
        for pc, (code, yes, no, _) in enumerate(rows):
            if code in (0x15, 0x45):
                self.assertLess(pc + 1 + yes, len(rows))
                self.assertLess(pc + 1 + no, len(rows))

    def test_ctypes_admission_uses_owned_pointers_and_fixed_resource_pairs(self):
        calls = []
        def prctl(option, argument, pointer, last, final):
            calls.append((option, argument, pointer, last, final))
            if option == 22:
                descriptor = child.SockFprog.from_address(pointer)
                self.assertEqual(descriptor.length, len(child.filter_instructions()))
                for index, row in enumerate(child.filter_instructions()):
                    item = descriptor.filter[index]
                    self.assertEqual((item.code, item.jt, item.jf, item.k), row)
            return 0
        function = Mock(side_effect=prctl)
        pairs = [value for _, value in child.limits().values()]
        with patch.object(child.sys, "platform", "linux"), patch.object(child.platform, "machine", return_value="aarch64"), \
             patch.object(child.platform, "release", return_value="fixture"), \
             patch("builtins.open", return_value=io.BytesIO(b"trusted runtime")), \
             patch.object(child.os, "listdir", return_value=[]), \
             patch.object(child.resource, "setrlimit") as setter, \
             patch.object(child.resource, "getrlimit", side_effect=[(v, v) for v in pairs]), \
             patch.object(child.ctypes, "CDLL", return_value=SimpleNamespace(prctl=function)):
            admitted, runtime = child.enter()
        self.assertEqual(calls[0], (38, 1, 0, 0, 0))
        self.assertEqual(calls[1][:2], (22, 2))
        self.assertEqual(setter.call_count, 5)
        self.assertEqual(admitted["RLIMIT_CPU"], [4, 4])
        self.assertEqual(runtime["python_sha256"], hashlib.sha256(b"trusted runtime").hexdigest())

    def test_failed_create_cleans_only_its_proven_owned_container(self):

    def fake_capture(self, stream_name, behavior):
        class Stream(io.BytesIO):
            def fileno(self): return 10 if self is stdout else 11
        class Selector:
            def __init__(self): self.keys = {}
            def register(self, stream, _, name): self.keys[stream] = SimpleNamespace(fileobj=stream, data=name)
            def unregister(self, stream): del self.keys[stream]
            def get_map(self): return self.keys
            def select(self, _):
                wanted = "stdout" if state["reads"] == 0 else stream_name
                return [(next(key for key in self.keys.values() if key.data == wanted), 1)]
            def close(self): self.keys.clear()
        stdout, stderr = Stream(), Stream()
        code = -signal.SIGKILL if behavior in ("flood", "wall") else 0
        worker = SimpleNamespace(pid=123, stdout=stdout, stderr=stderr, returncode=code, wait=lambda **_: code)
        mode = "wall_timeout" if behavior == "wall" else stream_name + "_flood"
        initial = (json.dumps(ready(mode)) + "\n").encode()
        state = {"reads": 0, "clock": 0.0}
        def read(_, size):
            self.assertEqual(size, 4096)
            state["reads"] += 1
            if state["reads"] == 1:
                if behavior == "wall": state["clock"] = 7.0
                return initial
            return b"x" * 4096
        with patch.object(probe.selectors, "DefaultSelector", Selector), \
             patch.object(probe.subprocess, "Popen", return_value=worker), \
             patch.object(probe.os, "set_blocking", side_effect=OSError("authored setup failure") if behavior == "setup" else None), \
             patch.object(probe.os, "read", side_effect=read), \
             patch.object(probe.time, "monotonic", side_effect=lambda: state["clock"]), \
             patch.object(probe, "signal_owned_group", return_value=behavior != "setup"), \
             patch.object(probe, "live_group_members", return_value=[]):
            item = probe.capture(["owned-fixture"], 6, 65536)
        return item, mode

    def test_transport_caps_both_pipes_and_wall_without_running_a_child(self):
        for stream, behavior in [("stdout", "flood"), ("stderr", "flood"), ("stdout", "wall")]:
            transport, mode = self.fake_capture(stream, behavior)
            expected = capture(mode, container_code=137, transport_exit_code=-signal.SIGKILL,
                               execution_killed=True, container_forced_kill=True)
            expected.update(transport)
            expected["transport_exit_code"] = transport["exit_code"]
            expected["exit_code"] = -signal.SIGKILL
            probe.check_record(expected)
            if behavior == "flood":
                self.assertEqual(transport["stdout_captured_bytes"] + transport["stderr_captured_bytes"], 65536)
            else: self.assertTrue(transport["wall_timed_out"])
