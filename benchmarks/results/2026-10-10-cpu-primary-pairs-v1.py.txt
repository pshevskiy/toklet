"""Sequential same-binary primary comparison; preserve the idle scalar batch."""
import hashlib
import json
import os
from pathlib import Path
import signal
import subprocess
import sys
import time
from datetime import datetime, timezone

ROOT = Path('/Users/coloner/work/rust/toklet')
DRIVER, NUMERICAL = 17695, 17701
RECEIPT = ROOT/'benchmarks/results/2026-10-10-cpu-tiled-qualification-v1.json'
OUTPUT = ROOT/'benchmarks/raw/2026-10-10-cpu-primary-pairs-v1.json'

def stamp(): return datetime.now(timezone.utc).isoformat()
def digest(path): return hashlib.sha256(path.read_bytes()).hexdigest()
def process(pid):
    output = subprocess.check_output(['/bin/ps','-o','pid=,ppid=,stat=,%cpu=,time=,command=','-p',str(pid)], text=True).strip()
    fields = output.split(None,5)
    if len(fields)!=6: raise RuntimeError(f'Owned process unavailable: {pid}')
    return {'pid':int(fields[0]),'ppid':int(fields[1]),'state':fields[2], 'cpu':float(fields[3]),'cpu_time':fields[4],'command':fields[5]}
def boundary():
    driver, worker = process(DRIVER), process(NUMERICAL)
    if driver['ppid']!=17694 or 'tools/bench/cpu.py --workload boundary' not in driver['command'] or not driver['state'].startswith('T'):
        raise RuntimeError('Scalar transport is not held at its owned boundary')
    if worker['ppid']!=DRIVER or worker['command']!=str(ROOT/'target/release/examples/cpu_bench')+' '+str(ROOT/'models/Qwen2.5-0.5B-Instruct') or not worker['state'].startswith('S') or worker['cpu']>0.1:
        raise RuntimeError('Frozen scalar numerical worker is not idle')
    return {'driver':driver,'numerical':worker}
def save(record):
    tmp = OUTPUT.with_suffix('.json.pending')
    tmp.write_text(json.dumps(record,indent=2,sort_keys=True,allow_nan=False)+'\n')
    tmp.replace(OUTPUT)

if OUTPUT.exists() or OUTPUT.with_suffix('.json.pending').exists(): raise RuntimeError('Refusing to overwrite orchestration evidence')
state = boundary()
time.sleep(1)
confirmed = boundary()
if state['numerical']['cpu_time']!=confirmed['numerical']['cpu_time']: raise RuntimeError('Scalar numerical worker was active')
record={'format':'toklet-cpu-primary-series-v1','complete':False,'started_utc':stamp(),
        'primary_metric':'warm 128-input/128-output TTFT','order':['scalar','tiled','tiled','scalar'],
        'qualification_sha256':digest(RECEIPT),'driver_sha256':digest(ROOT/'tools/bench/cpu.py'),
        'orchestrator_sha256':digest(Path(__file__)),'baseline_before':confirmed,
        'scope':'Separate complete 3-warmup/10-sample runs from the same qualified binary; order reversed in the confirming pair. Scalar batch numerical process remains idle. Full matrix acceptance remains pending.', 'runs':[]}
save(record)
try:
    for index,(pair,kernel) in enumerate(((1,'scalar'),(1,'tiled'),(2,'tiled'),(2,'scalar'))):
        output=ROOT/f'benchmarks/raw/2026-10-10-cpu-primary-pair-{pair}-{kernel}.json'
        command=[sys.executable,'tools/bench/cpu.py','--workload','short','--cpu-projection',kernel,'--cpu-qualification',str(RECEIPT),'--output',str(output)]
        item={'pair':pair,'projection':kernel,'started_utc':stamp(),'command':command,'output':str(output.relative_to(ROOT))}
        record['runs'].append(item); save(record)
        print(f'Primary pair {pair}, {kernel}: started {item["started_utc"]}',flush=True)
        child = subprocess.Popen(command,cwd=ROOT,start_new_session=True)
        try:
            code = child.wait(timeout=3*60*60)
        except BaseException:
            try: os.killpg(child.pid,signal.SIGTERM)
            except ProcessLookupError: pass
            try: child.wait(timeout=5)
            except subprocess.TimeoutExpired:
                try: os.killpg(child.pid,signal.SIGKILL)
                except ProcessLookupError: pass
                child.wait(timeout=5)
            # Reap any numerical descendant after abrupt driver termination.
            try: os.killpg(child.pid,signal.SIGKILL)
            except ProcessLookupError: pass
            raise
        if code:
            raise subprocess.CalledProcessError(code,command)
        item.update(finished_utc=stamp(),sha256=digest(output),summary=json.loads(output.read_text())['summary'])
        observed=boundary()
        if observed['numerical']['cpu_time']!=confirmed['numerical']['cpu_time']: raise RuntimeError('Scalar numerical activity overlapped comparison')
        item['baseline_after']=observed
        save(record)
    record['complete']=True
except BaseException as error:
    record['failure']=f'{type(error).__name__}: {error}'
    raise
finally:
    record['finished_utc']=stamp()
    try:
        record['baseline_final']=boundary()
    finally:
        current=process(DRIVER)
        if current['ppid']==17694 and 'tools/bench/cpu.py --workload boundary' in current['command'] and current['state'].startswith('T'):
            os.kill(DRIVER,signal.SIGCONT)
            record['scalar_transport_resumed_utc']=stamp()
        save(record)
