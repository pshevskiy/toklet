"""Validate only small saved medium records; never execute or hash a worker/model."""
import hashlib
import json
import math
from pathlib import Path
import statistics

from jsonschema import Draft202012Validator

ROOT = Path(__file__).resolve().parents[2]
RESULTS = ROOT / "benchmarks/results"

def read(name):
    return json.loads((ROOT / name).read_text())

def digest(name):
    return hashlib.sha256((ROOT / name).read_bytes()).hexdigest()

result_name = "benchmarks/results/2026-10-10-cpu-medium.json"
raw_name = result_name + ".samples.jsonl"
result = read(result_name)
Draft202012Validator(read("benchmarks/result.schema.json")).validate(result)
rows = [json.loads(line) for line in (ROOT / raw_name).read_text().splitlines()]
assert len(rows) == len(result["samples"]) == 10
assert result["setup"]["warmup_repetitions"] == 3
assert result["runtime"] == "Toklet scalar CPU"
assert result["representation"] == {"group_size": None, "kv_dtype": "FP32", "packed_format_version": None, "weight_dtype": "FP32"}
assert result["identity"]["engine_commit"] == "8a5ebc8ba320d073cbb4d1f2db8cbded22e0760f"
assert result["identity"]["working_tree_dirty"] is False
assert result["workload"] == {
    "cache_state": "empty", "context_capacity": 2048,
    "continuation_sha256": "29fc3d4bc6b96644867b4d8df0c30f0009d814edf6e5579aa02d8f16ebedec07",
    "name": "medium_teacher_forced_128", "output_steps": 128,
    "prompt_sha256": "4ff74dae2ee5b778f9313f4c707f9ccb6415609a9c0653684cce7be63d457dd6",
    "prompt_tokens": 512, "stop_policy": "fixed_length_teacher_forced_ignore_eos",
    "used_context_before_request": 0,
}
contract = read("benchmarks/contract-v1/manifest.json")
for key, item in [("prompt_512", "prompt_sha256"), ("continuation_128", "continuation_sha256")]:
    array = contract["arrays"][key]
    assert digest("benchmarks/contract-v1/" + array["file"]) == array["sha256"] == result["workload"][item]
for row, sample in zip(rows, result["samples"]):
    assert row["sample"] == sample
    state = row["state"]
    assert (state["reused_tokens"], state["prefill_tokens"], state["evaluated_tokens"], state["pending_tokens"], state["emitted_tokens"]) == (0, 512, 639, 1, 128)
    for key in ("prefill_seconds", "ttft_seconds", "decode_seconds", "total_seconds", "token_event_seconds", "emitted_tokens"):
        assert state[key] == sample[key]
    events = sample["token_event_seconds"]
    assert len(events) == 128 and events[0] == sample["ttft_seconds"]
    assert all(math.isfinite(t) and t >= 0 for t in events)
    assert all(b >= a for a, b in zip(events, events[1:]))
    assert math.isclose(sample["decode_seconds"], events[-1] - events[0], rel_tol=0, abs_tol=1e-9)
    assert 0 <= sample["prefill_seconds"] <= events[0] <= events[-1] <= sample["total_seconds"]
    assert sample["first_visible_text_seconds"] is None and sample["gpu_peak_allocation_bytes"] is None
    assert sample["process_peak_footprint_bytes"] < 6 * 1024**3
samples = result["samples"]
summary = {
    "ttft_median_seconds": statistics.median(s["ttft_seconds"] for s in samples),
    "ttft_min_seconds": min(s["ttft_seconds"] for s in samples),
    "ttft_max_seconds": max(s["ttft_seconds"] for s in samples),
    "request_median_seconds": statistics.median(s["total_seconds"] for s in samples),
    "decode_tokens_per_second_median": statistics.median(127 / (s["token_event_seconds"][-1] - s["token_event_seconds"][0]) for s in samples),
    "prefill_tokens_per_second_median": statistics.median(512 / s["prefill_seconds"] for s in samples),
}
assert summary == result["summary"]
assert max(s["process_peak_footprint_bytes"] for s in samples) == result["memory"]["peak_process_bytes"] == 2043840528
assert result["memory"]["logical_weights_bytes"] == 1976131072
assert result["memory"]["allocated_kv_bytes"] == 50331648
assert result["memory"]["scratch_bytes"] == 10436608
assert "Admitted process estimate 2172348800 bytes; chunk 128; fixed capacity 2048;" in result["notes"][1]
driver_name = "benchmarks/results/2026-10-10-cpu-medium-source-driver-v1.py.txt"
assert digest(driver_name) == "887436019c6f634eff60ba8a6ba73e22452d4550bed1115f4a75a46fa483a21a"
window_name = "benchmarks/results/2026-10-10-cpu-qwen-silu-check-window-completed.json"
window = read(window_name)
assert window["completed"] is True and window["working_tree_clean_at_verification"] is True
assert window["reviewed_commits"] == ["706edb5", "95733e4"]
assert window["frozen_binaries"]["target/release/examples/cpu_bench"] == "e3debabbae7db9e5801bc9bfba177275444ebe1d9849eb3ad0f2dff487ac8dea"
for name in [result_name, raw_name]:
    assert digest(name) == window["records"][name.replace("benchmarks/results/", "benchmarks/raw/")]
hold = read("benchmarks/results/2026-10-10-cpu-qwen-silu-check-window-hold.json")
idle = read("benchmarks/results/2026-10-10-cpu-qwen-silu-check-window-idle.json")
assert hold["active_numerical_worker_pid"] == idle["exited_numerical_worker_pid"] == 54916
assert idle["completed_workload"] == "medium" and idle["measured_requests"] == 10
assert idle["idle_verified_utc"] < window["verified_utc"] < window["orchestration_resumed_utc"]
bound = [result_name, raw_name, driver_name, window_name,
         "benchmarks/results/2026-10-10-cpu-qwen-silu-check-window-hold.json",
         "benchmarks/results/2026-10-10-cpu-qwen-silu-check-window-idle.json"]
analysis = {
    "format": "toklet-saved-cpu-medium-analysis-v1", "completed": True,
    "scope": "Small saved-evidence validation only; no native execution, model reads, or binary hashes. Prior idle-window proof binds those identities.",
    "checks": ["schema", "ten exact raw/sample pairs", "state and token events", "independent summary", "fixed input hashes", "memory totals", "historical driver", "idle-window identity and ordering"],
    "summary": summary,
    "requests_seconds": [s["total_seconds"] for s in samples],
    "request_min_seconds": min(s["total_seconds"] for s in samples),
    "request_max_seconds": max(s["total_seconds"] for s in samples),
    "records_sha256": {name: digest(name) for name in bound},
    "producer_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
}
out = RESULTS / "2026-10-10-cpu-medium-analysis-v1.json"
with out.open("x") as stream:
    json.dump(analysis, stream, indent=2, allow_nan=False)
    stream.write("\n")
print(json.dumps({"completed": True, "summary": summary, "request_min_seconds": analysis["request_min_seconds"], "request_max_seconds": analysis["request_max_seconds"]}))
