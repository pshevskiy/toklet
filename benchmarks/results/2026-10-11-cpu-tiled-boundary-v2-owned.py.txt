"""Record only the verified boundary launcher's three owned processes."""
from datetime import datetime, timezone
from pathlib import Path
import hashlib, json, subprocess

root = Path.cwd()
pids = [50988, 51014, 51022]
command = ['ps', '-p', ','.join(map(str, pids)), '-o', 'pid=,ppid=,pgid=,stat=,command=']
p = subprocess.run(command, text=True, capture_output=True)
assert p.returncode == 0 and not p.stderr
rows = {int(row[0]): row for line in p.stdout.splitlines() if len(row := line.split(None, 4)) == 5}
assert set(rows) == set(pids)
assert [(int(rows[pid][1]), int(rows[pid][2])) for pid in pids] == [(35939, 50988), (50988, 50988), (51014, 50988)]
assert all(row[3][0] in ('R', 'S', 'I') and not any(c in row[3] for c in ('T', 't', 'Z')) for row in rows.values())
python_image = '/Library/Frameworks/Python.framework/Versions/3.12/Resources/Python.app/Contents/MacOS/Python'
assert rows[50988][4] == python_image+' artifacts/launch-tiled-boundary-v2.py --workload boundary'
expected_driver = python_image+' tools/bench/cpu.py --workload boundary --cpu-projection tiled --cpu-qualification '+str(root/'benchmarks/results/2026-10-10-cpu-tiled-qualification-v1.json')+' --output benchmarks/raw/2026-10-11-cpu-tiled-boundary-v2.json'
assert rows[51014][4] == expected_driver
assert rows[51022][4] == str(root/'artifacts/cpu-tiled-engine-v1/release/examples/cpu_bench')+' '+str(root/'models/Qwen2.5-0.5B-Instruct')+' 128 tiled'
launch = root/'benchmarks/raw/2026-10-11-cpu-tiled-boundary-v2-launch.json'
record = {'format': 'toklet-cpu-tiled-workload-owned-v1', 'session_id': 13791,
          'observed_utc': datetime.now(timezone.utc).isoformat(),
          'launcher_pid': 50988, 'driver_pid': 51014, 'numerical_worker_pid': 51022,
          'owned_process_group': 50988, 'snapshot': p.stdout,
          'launch_receipt_sha256': hashlib.sha256(launch.read_bytes()).hexdigest(),
          'producer_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
with (root/'artifacts/2026-10-11-cpu-tiled-boundary-v2-owned.json').open('x') as stream:
    json.dump(record, stream, indent=2); stream.write('\n')
print(json.dumps(record))
