"""Inactive check producer; only after scalar collection exit and evidence commit.

The four grading/window modules must first be promoted to tools/bench. These
checks execute offline fixtures, formatting, Clippy and Rust tests, not model
inference or generated programs. The qualified Linux VM stays stopped.
"""
from datetime import datetime, timezone
import json
import os
from pathlib import Path
import sys

sys.path.insert(0, 'tools/bench')
import held_out_tasks as collector
import held_out_task_grading as grader
import prepare
import quality

ROOT = Path.cwd()
OUTPUT = ROOT / 'benchmarks/raw/2026-10-11-held-out-task-grading-checks-v1.json'
PROOF = ROOT / 'benchmarks/results/2026-10-11-held-out-task-scalar-idle-v1.json'
NAMES = ('held_out_task_grading.py', 'test_held_out_task_grading.py',
         'held_out_task_window.py', 'test_held_out_task_window.py')


def run():
    assert not OUTPUT.exists()
    identities = {'tools/bench/' + name: prepare.digest(ROOT / 'tools/bench' / name) for name in NAMES}
    record = {'format': 'toklet-held-out-task-grading-checks-v1', 'completed': False,
              'started_utc': datetime.now(timezone.utc).isoformat(),
              'producer_sha256': prepare.digest(Path(__file__)), 'source_sha256': identities,
              'admission': collector.idle(PROOF), 'checks': [],
              'scope': 'Offline grading/window checks only. No model, generated code, VM startup or task score.'}
    os.environ['CARGO_TARGET_DIR'] = str(ROOT / 'artifacts/cpu-tiled-cli-v1')
    record['check_target_directory'] = os.environ['CARGO_TARGET_DIR']
    try:
        collection_path = ROOT / 'benchmarks/results/2026-10-11-held-out-task-scalar-v1.json'
        collection, cases = grader.validate_collection(collection_path)
        assert collection['projection'] == 'scalar' and len(cases) == 100
        record['collection_validation'] = {
            'collection_sha256': prepare.digest(collection_path), 'projection': 'scalar', 'cases': len(cases),
            'scope': 'Real completed collection provenance/response validation only; no scoring or program execution.'}
        for command in [[str(ROOT / '.venv/bin/python'), '-m', 'unittest', 'discover',
                         '-s', 'tools/bench', '-p', 'test_*.py'],
                        ['cargo', 'fmt', '--check'],
                        ['cargo', 'clippy', '--locked', '--offline', '--all-targets', '--', '-D', 'warnings'],
                        ['cargo', 'test', '--locked', '--offline']]:
            quality.checked(command, record)
        assert identities == {name: prepare.digest(ROOT / name) for name in identities}
        record['end_admission'] = collector.idle(PROOF)
        record['completed'] = True
    finally:
        record['finished_utc'] = datetime.now(timezone.utc).isoformat()
        with OUTPUT.open('x') as stream:
            json.dump(record, stream, indent=2, allow_nan=False)
            stream.write('\n')
    print('Complete held-out grading offline/format/clippy/Rust checks passed', flush=True)


if __name__ == '__main__':
    run()
