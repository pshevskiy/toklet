def read(name):

def digest(path):
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()

def independently_replay(event, source, expected_anchors):
    commands = active = guards = 0
    fraction = absolute = 0.0
    assert len(event["result"]["cases"]) == len(source)
    for case, item in zip(event["result"]["cases"], source):
        assert case["id"] == item["id"]
        anchor = expected_anchors[item["id"]]
        count = item["rows"] * item["width"]
        expected = []
        # Independent replay does not call the product helper. Half significands
        # multiply exactly; additionally prove exact FP32 storage for each product.
        for a, b in zip(item["gate_bits"], item["up_bits"]):
            left = struct.unpack("<e", struct.pack("<H", a))[0]
            right = struct.unpack("<e", struct.pack("<H", b))[0]
            product = left * right
            assert struct.unpack("<f", struct.pack("<f", product))[0] == product
            expected.append(struct.unpack("<H", struct.pack("<e", product))[0])
        assert len(expected) == count and expected == anchor["bits"]
        actual = struct.unpack(f"<{count}e", struct.pack(f"<{count}H", *expected))
        for row in case["rows"]:
            assert row["output_bits"] == expected + [0x55a1] * 7
            for value, reference in zip(actual, anchor["reference"]):
                assert math.isfinite(value) and math.isfinite(reference)
                error = abs(value - reference)
                absolute = max(absolute, error)
                if anchor["gate"] is not None:
                    gate = anchor["gate"]
                    used = error / (gate["atol"] + gate["rtol"] * abs(reference))
                    assert used <= 1
                    fraction = max(fraction, used)
                else:
                    assert error == 0
            commands += 1
            active += count
            guards += 7
    return commands, active, guards, fraction, absolute
