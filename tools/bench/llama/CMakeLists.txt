cmake_minimum_required(VERSION 3.14)
project(toklet_llama_bench C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
if(NOT DEFINED LLAMA_SOURCE)
    message(FATAL_ERROR "Set LLAMA_SOURCE to the contract-pinned local source")
endif()
add_subdirectory(${LLAMA_SOURCE} ${CMAKE_BINARY_DIR}/llama)
add_executable(toklet-llama-bench worker.cpp)
target_include_directories(toklet-llama-bench PRIVATE ${LLAMA_SOURCE}/vendor)
target_link_libraries(toklet-llama-bench PRIVATE llama)
target_compile_options(toklet-llama-bench PRIVATE -Wall -Wextra -Werror)
